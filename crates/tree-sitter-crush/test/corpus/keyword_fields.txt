// (CRUSH-367, GitHub issue #367) is listed by `crush-run caps` and must parse.
// `match` stays a keyword everywhere else (match expressions, bare calls).

    print(regex.match("^[a-z]+", "ab cd"))
    let m = regex.match("x", "y")
}
